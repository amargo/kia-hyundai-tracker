# Kia Hyundai Tracker - Cronjob Examples
# Replace /path/to/your/project with your actual project path
# Replace /path/to/python with your Python executable path

# Basic vehicle data refresh every 15 minutes
*/15 * * * * cd /path/to/your/project && /path/to/python main.py --action refresh

# Process trips every hour
0 * * * * cd /path/to/your/project && /path/to/python main.py --action trips --verbose

# Save daily statistics once per day at 23:30
30 23 * * * cd /path/to/your/project && /path/to/python main.py --action daily_stats --verbose

# Full refresh (all data) once per day at 06:00
0 6 * * * cd /path/to/your/project && /path/to/python main.py --action all --verbose
